# EC6108V9C / Hi3798MV100 单板叠加层
#
# 叠加方法只用内核源码树里的工具，不依赖任何外部脚本：
#
#   FRAG=mv100.config
#   OUT=/tmp/out.config
#
#   # 1) 基线：发行版的 config（/boot/config-*），或
#   #    make ARCH=arm multi_v7_defconfig 的产物
#   BASE=/path/to/debian.config
#
#   # 2) 瘦身（可选，纯机械）：只把 =m 行改成 =n，不碰任何 =y 行，
#   #    所以基线的调试与安全姿态原样保留；不瘦身就直接 cp $BASE $OUT
#   sed -E 's/^(CONFIG_[A-Za-z0-9_]+)=m$/\1=n/' $BASE > $OUT
#
#   # 3) 叠加本文件（-m 表示只合并、不跑 make）
#   KCONFIG_CONFIG=$OUT ./scripts/kconfig/merge_config.sh -m $OUT $FRAG
#
#   # 4) 让 kconfig 补全依赖
#   make ARCH=arm CROSS_COMPILE=arm-linux-gnueabihf- \
#       KCONFIG_CONFIG=$OUT olddefconfig
#
#   # 5) 自检：改名或被依赖挡住造成的失效是静默的，只能逐行确认
#   grep '^CONFIG_' $FRAG > /tmp/board.want
#   while read -r line; do
#       grep -qxF "$line" $OUT || echo "没生效：$line"
#   done < /tmp/board.want
#
# 基线前提：ARCH_MULTI_V7=y。它关着时 ARCH_HISI 那行会被依赖挡住，
# 下面整段本设备硬件会一起静默消失。
#
# 只写 olddefconfig 推不出来的行：被 select 拉起的（DMA_ENGINE、SND_PCM、
# FS_ENCRYPTION_ALGS、CRYPTO_XTS、RESET_CONTROLLER 等）和 default ARCH_HISI
# 自动打开的（HISI_THERMAL、HW_RANDOM_HISTB、其它 hisi 时钟驱动）一律不写，
# 内核换代后它们自己会回来。
# 例外：CONFIG_VIDEO_DEV 有 default（MEDIA_*_SUPPORT 里任意为 y 就指向 y），但基线
# 已经把它钉成 m，olddefconfig 对已存在的合法取值一律保留 m。不写这一行
# VIDEO_HISTB_VDEC 只会落进 m——步骤 5 的自检实测到并报"没生效"。
#
# 书写顺序有讲究：kconfig 按行顺序算可见性，父符号必须写在子符号之前
# （先 USB/USB_EHCI_HCD，后 USB_EHCI_HCD_PLATFORM），否则子符号的 y 落不进去。

# TRNG 没有手写行：HW_RANDOM_HISTB 是 default ARCH_HISI，olddefconfig 自己给 y。
# 板上实测该块起来就能出数（dtb 节点 rng@8005000，4 KiB 窗口），门位在 PERI CRG
# +0xc4 的 bit0/bit4，与 SHA 引擎共用，引导程序已经把钟开着，所以驱动不需要
# clocks/resets 属性。
# I2C 控制器在片内。板 dts 里三条总线都使能了，板上 i2c-1/2/3 三个 adapter 注册
# 成功；但门位 CRG27 读回 0、clk_i2c0/1/2 的 prepare/enable 都是 0 ⇒ 只到注册，
# 没做过一次真实传输，复位位也没人释放（驱动不用 reset 框架，原厂 dts 也不带
# reset 属性）。EDID 仍然走自建 DDC，不依赖这几条。

# 这是叠加层里唯一一行姿态覆盖：基线给的是 DEBUG_INFO=y + DWARF_TOOLCHAIN_DEFAULT=y，
# vmlinux 变大、编译变慢。
#
# 必须写 choice 成员名：DEBUG_INFO 本身没有 prompt（lib/Kconfig.debug 里是纯
# bool），写 CONFIG_DEBUG_INFO=n 会被 choice 里仍为 y 的成员 select 回来，第 5 步
# 逐行自检也看不见它。DEBUG_INFO_NONE 只 depends on DEBUG_KERNEL（基线已是 y），
# 不像 DWARF_* 那几个成员带 AS_VERSION/CC_IS_CLANG 探测依赖。
# 实测形态见同基线同叠加层跑出的 debug.config（CONFIG_DEBUG_INFO_NONE=y、三个
# DWARF_* 全部 is not set、无 CONFIG_DEBUG_INFO 行）。
CONFIG_DEBUG_INFO_NONE=y
